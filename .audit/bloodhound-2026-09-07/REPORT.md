with players(name, created_at, plays) as (values
  ('NY',   timestamptz '2026-09-01 19:00Z', array[timestamptz '2026-09-01 19:05Z', timestamptz '2026-09-02 02:00Z']),
  ('Oslo', timestamptz '2026-09-01 20:00Z', array[timestamptz '2026-09-01 22:30Z'])),
runs as (select generate_series(timestamptz '2026-09-02 00:45Z', timestamptz '2026-09-03 23:45Z', interval '1 hour') as run_at),
ev as (select p.name, r.run_at,
    p.created_at >= ((r.run_at at time zone 'UTC')::date - 1)::timestamptz and p.created_at < ((r.run_at at time zone 'UTC')::date)::timestamptz as in_signup_window,
    exists (select 1 from unnest(p.plays) t where t >= ((r.run_at at time zone 'UTC')::date - 1)::timestamptz and t < ((r.run_at at time zone 'UTC')::date)::timestamptz) as played_yesterday,

  from players p cross join runs r)
select name, min(run_at) filter (where in_signup_window and played_yesterday and not played_today) as first_selected_run,
       count(*) filter (where in_signup_window and played_yesterday and not played_today) as selected_runs
from ev group by name;
